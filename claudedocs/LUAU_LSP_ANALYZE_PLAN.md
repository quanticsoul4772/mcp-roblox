# Implementation Plan: luau-lsp analyze MCP Tool

## Overview

Add `luau_lsp_analyze` MCP tool for static type-checking Luau code before syncing to Roblox Studio. This tool wraps the `luau-lsp analyze` CLI to provide structured diagnostic output.

### Value Proposition
- **Prevents "sync to Studio, see red errors, fix, repeat" loop**
- Catches type mismatches, undefined variables, deprecated APIs
- JSON output integrates cleanly with MCP responses
- Works with existing Rojo sourcemaps already generated by `rojo_sourcemap`

### Implementation Complexity: LOW
- Single CLI call pattern (same as Selene, StyLua, Lune)
- Follows established trait-based abstraction pattern
- Existing toolchain patterns provide clear template

```rust
//! Luau-LSP static analysis integration
//!
//! Provides type checking for Luau scripts using luau-lsp analyze command.

use async_trait::async_trait;
use serde::{Deserialize, Serialize};
use std::path::Path;
use std::process::Stdio;
use tokio::process::Command;

use crate::error::RobloxMcpError;
use crate::tools::timeout::execute_with_timeout;

/// Result from analyzing Luau scripts
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AnalyzeResult {
    /// Root path that was analyzed
    pub path: String,
    /// List of diagnostics found
    pub diagnostics: Vec<AnalyzeDiagnostic>,
    /// Total error count
    pub error_count: usize,
    /// Total warning count
    pub warning_count: usize,
    /// Files analyzed count
    pub files_analyzed: usize,
}

/// Individual diagnostic from luau-lsp analyze
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AnalyzeDiagnostic {
    /// Severity: "Error", "Warning", "Information", "Hint"
    pub severity: String,
    /// Diagnostic code (e.g., "TypeError", "UnknownGlobal")
    pub code: String,
    /// Human-readable message
    pub message: String,
    /// File path where diagnostic occurred
    pub file: String,
    /// Start line (1-indexed)
    pub start_line: u32,
    /// Start column (1-indexed)
    pub start_column: u32,
    /// End line (1-indexed)
    #[serde(default)]
    pub end_line: Option<u32>,
    /// End column (1-indexed)
    #[serde(default)]
    pub end_column: Option<u32>,
}

// ============================================================================
// Trait Abstraction
// ============================================================================

#[async_trait]
pub trait LuauLspRunner: Send + Sync {
    /// Analyze Luau scripts for type errors and warnings
    ///
    /// # Arguments
    /// * `path` - Path to file or directory to analyze
    /// * `sourcemap_path` - Optional path to Rojo sourcemap.json
    /// * `definitions` - Optional paths to definition files (API types)
    async fn analyze(
        &self,
        path: &Path,
        sourcemap_path: Option<&Path>,
        definitions: &[&Path],
    ) -> Result<AnalyzeResult, RobloxMcpError>;
}

/// Production runner using luau-lsp CLI
#[derive(Debug, Default, Clone)]
pub struct DefaultLuauLspRunner;

impl DefaultLuauLspRunner {
    pub fn new() -> Self {
        Self
    }
}

#[async_trait]
impl LuauLspRunner for DefaultLuauLspRunner {
    async fn analyze(
        &self,
        path: &Path,
        sourcemap_path: Option<&Path>,
        definitions: &[&Path],
    ) -> Result<AnalyzeResult, RobloxMcpError> {
        analyze_path(path, sourcemap_path, definitions).await
    }
}

// ============================================================================
// Implementation
// ============================================================================

async fn analyze_path(
    path: &Path,
    sourcemap_path: Option<&Path>,
    definitions: &[&Path],
) -> Result<AnalyzeResult, RobloxMcpError> {
    let mut cmd = Command::new("luau-lsp");
    cmd.arg("analyze");

    // Use JSON formatter for structured output
    cmd.arg("--formatter=json");

    // Add sourcemap if provided (enables proper require resolution)
    if let Some(sourcemap) = sourcemap_path {
        cmd.arg(format!("--sourcemap={}", sourcemap.display()));
    }

    // Add definition files
    for def in definitions {
        cmd.arg(format!("--definitions={}", def.display()));
    }

    // Add the target path
    cmd.arg(path);

    cmd.stdout(Stdio::piped()).stderr(Stdio::piped());

    let output = execute_with_timeout(cmd, "luau-lsp", None)
        .await
        .map_err(|e| {
            if e.to_string().contains("timed out") {
                e
            } else {
                RobloxMcpError::ToolNotInstalled {
                    tool: "luau-lsp".to_string(),
                    install_hint: "Install via: aftman add johnnymorganz/luau-lsp".to_string(),
                }
            }
        })?;

    parse_luau_lsp_output(&output.stdout, &output.stderr, path)
}

/// Parse luau-lsp JSON output into AnalyzeResult
pub fn parse_luau_lsp_output(
    stdout: &[u8],
    stderr: &[u8],
    path: &Path,
) -> Result<AnalyzeResult, RobloxMcpError> {
    let stdout_str = String::from_utf8_lossy(stdout);
    let stderr_str = String::from_utf8_lossy(stderr);

    // If stderr has content and stdout is empty, luau-lsp failed
    if stdout_str.is_empty() && !stderr_str.is_empty() {
        return Err(RobloxMcpError::ConfigError(format!(
            "luau-lsp error: {}",
            stderr_str.trim()
        )));
    }

    // Parse JSON output (format TBD - may need adjustment based on actual output)
    let mut diagnostics = Vec::new();
    let mut files_analyzed = 0;

    // luau-lsp may output one JSON object per file or per diagnostic
    for line in stdout_str.lines() {
        if line.trim().is_empty() {
            continue;
        }

        // Try to parse as diagnostic array or single diagnostic
        if let Ok(diags) = serde_json::from_str::<Vec<LuauLspDiagnostic>>(line) {
            for diag in diags {
                diagnostics.push(convert_diagnostic(diag));
            }
            files_analyzed += 1;
        } else if let Ok(diag) = serde_json::from_str::<LuauLspDiagnostic>(line) {
            diagnostics.push(convert_diagnostic(diag));
        }
    }

    let error_count = diagnostics.iter().filter(|d| d.severity == "Error").count();
    let warning_count = diagnostics.iter().filter(|d| d.severity == "Warning").count();

    Ok(AnalyzeResult {
        path: path.display().to_string(),
        diagnostics,
        error_count,
        warning_count,
        files_analyzed,
    })
}

// Internal struct matching luau-lsp JSON format (to be verified)
#[derive(Debug, Deserialize)]
struct LuauLspDiagnostic {
    severity: Option<u8>,  // LSP severity: 1=Error, 2=Warning, 3=Info, 4=Hint
    message: String,
    range: LuauLspRange,
    source: Option<String>,
    code: Option<String>,
}

#[derive(Debug, Deserialize)]
struct LuauLspRange {
    start: LuauLspPosition,
    end: LuauLspPosition,
}

#[derive(Debug, Deserialize)]
struct LuauLspPosition {
    line: u32,
    character: u32,
}

fn convert_diagnostic(diag: LuauLspDiagnostic) -> AnalyzeDiagnostic {
    let severity = match diag.severity.unwrap_or(1) {
        1 => "Error",
        2 => "Warning",
        3 => "Information",
        4 => "Hint",
        _ => "Unknown",
    }.to_string();

    AnalyzeDiagnostic {
        severity,
        code: diag.code.unwrap_or_else(|| "Unknown".to_string()),
        message: diag.message,
        file: diag.source.unwrap_or_default(),
        start_line: diag.range.start.line + 1,  // Convert 0-indexed to 1-indexed
        start_column: diag.range.start.character + 1,
        end_line: Some(diag.range.end.line + 1),
        end_column: Some(diag.range.end.character + 1),
    }
}

#[cfg(test)]
pub mod mock {
    use super::*;
    use std::collections::VecDeque;
    use std::sync::{Arc, Mutex};

    #[derive(Clone)]
    pub struct MockLuauLspRunner {

        responses: VecDeque<Result<AnalyzeResult, RobloxMcpError>>,
        calls: Vec<MockAnalyzeCall>,
    }

    #[derive(Debug, Clone)]
    pub struct MockAnalyzeCall {
        pub path: String,
        pub sourcemap_path: Option<String>,
        pub definitions: Vec<String>,
    }

    impl MockLuauLspRunner {
        pub fn new() -> Self {
            Self {
                state: Arc::new(Mutex::new(MockState {
                    responses: VecDeque::new(),
                    calls: Vec::new(),
                })),
            }
        }

        pub fn queue_response(&self, response: Result<AnalyzeResult, RobloxMcpError>) {
            self.state.lock().unwrap().responses.push_back(response);
        }

        pub fn calls(&self) -> Vec<MockAnalyzeCall> {
            self.state.lock().unwrap().calls.clone()
        }
    }

    impl Default for MockLuauLspRunner {
        fn default() -> Self {
            Self::new()
        }
    }

    impl std::fmt::Debug for MockLuauLspRunner {
        fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
            f.debug_struct("MockLuauLspRunner").finish()
        }
    }

    #[async_trait]
    impl LuauLspRunner for MockLuauLspRunner {
        async fn analyze(
            &self,
            path: &Path,
            sourcemap_path: Option<&Path>,
            definitions: &[&Path],
        ) -> Result<AnalyzeResult, RobloxMcpError> {
            let mut state = self.state.lock().unwrap();
            state.calls.push(MockAnalyzeCall {
                path: path.display().to_string(),
                sourcemap_path: sourcemap_path.map(|p| p.display().to_string()),
                definitions: definitions.iter().map(|p| p.display().to_string()).collect(),
            });

            state.responses.pop_front().unwrap_or_else(|| {
                Err(RobloxMcpError::ConfigError(
                    "MockLuauLspRunner: No response queued".into(),
                ))
            })
        }
    }
}
```

```rust
pub(crate) async fn luau_lsp_analyze_impl(
    &self,
    params: LuauLspAnalyzeParams,
) -> Result<CallToolResult, ErrorData> {
    let path = Path::new(&params.path);

    // Validate path is within project root
    let validated_path = validate_path(path, &self.project_root)
        .map_err(|e| ErrorData::internal_error(e.to_string(), None))?;

    // Parse optional sourcemap path
    let sourcemap_path = params
        .sourcemap_path
        .as_ref()
        .map(|p| Path::new(p).to_path_buf());

    // Parse definition paths
    let definitions: Vec<PathBuf> = params
        .definitions
        .unwrap_or_default()
        .iter()
        .map(|p| PathBuf::from(p))
        .collect();

    let def_refs: Vec<&Path> = definitions.iter().map(|p| p.as_path()).collect();

    // Run analysis
    let result = self
        .luau_lsp
        .analyze(&validated_path, sourcemap_path.as_deref(), &def_refs)
        .await
        .map_err(|e| ErrorData::internal_error(e.to_string(), None))?;

    Ok(CallToolResult::success(vec![Content::text(
        serde_json::to_string_pretty(&result)
            .map_err(|e| ErrorData::internal_error(e.to_string(), None))?,
    )]))
}
```
